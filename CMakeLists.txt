cmake_minimum_required(VERSION 3.18)
project(lr2021_sdr VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(lriq src/lriq.c src/lriq_fifo.c)
add_library(lr2021_iq src/lr2021_iq.c src/lr2021_iq_queue.c)
foreach(target lriq lr2021_iq)
  target_include_directories(${target} PUBLIC include)
  if(NOT MSVC)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
endforeach()
if(NOT MSVC)
  target_link_libraries(lriq PUBLIC m)
endif()
add_executable(lriq_tool tools/lriq_tool.c)
add_executable(lr2021_lab tools/lr2021_lab.c)
target_link_libraries(lriq_tool PRIVATE lriq)
if(NOT MSVC)
  target_compile_options(lriq_tool PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(lr2021_lab PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(lr2021_lab PRIVATE m)
endif()
if(BUILD_TESTING)
  add_executable(test_board_profile tests/test_board_profile.c firmware/esp32p4/main/board_profile.c)
  target_include_directories(test_board_profile PRIVATE firmware/esp32p4/main)
  if(NOT MSVC)
    target_compile_options(test_board_profile PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
  add_test(NAME board_profile COMMAND test_board_profile)
  add_executable(test_board_storage tests/test_board_storage.c firmware/esp32p4/main/board.c
    firmware/esp32p4/main/board_profile.c)
  target_include_directories(test_board_storage PRIVATE tests/board_stubs firmware/esp32p4/main)
  if(NOT MSVC)
    target_compile_options(test_board_storage PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
  add_test(NAME board_storage COMMAND test_board_storage)
  add_executable(test_tx_upload tests/test_tx_upload.c)
  add_test(NAME tx_upload_transactions COMMAND test_tx_upload)
  add_executable(test_tx_fifo tests/test_tx_fifo.c)
  add_test(NAME tx_fifo_refill COMMAND test_tx_fifo)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME tx_host_tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tx_tools.py)
    add_test(NAME tx_temperature_generator COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_temperature_tools.py)
    add_test(NAME tx_p25_tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_p25_tools.py)
  endif()
  add_executable(test_lriq tests/test_lriq.c)
  target_link_libraries(test_lriq PRIVATE lriq)
  add_test(NAME host_protocol COMMAND test_lriq)
  add_executable(test_radio tests/test_radio.c)
  target_link_libraries(test_radio PRIVATE lr2021_iq)
  add_test(NAME radio_transactions COMMAND test_radio)
  add_test(NAME offline_lab COMMAND lr2021_lab selftest)
  add_executable(test_queue tests/test_queue.c)
  target_link_libraries(test_queue PRIVATE lr2021_iq)
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(test_queue PRIVATE Threads::Threads)
  endif()
  add_test(NAME stream_queue COMMAND test_queue)
  set_tests_properties(stream_queue PROPERTIES TIMEOUT 20)
endif()
add_library(capture_example OBJECT examples/capture.c)
target_link_libraries(capture_example PRIVATE lr2021_iq lriq)
install(TARGETS lriq lr2021_iq lriq_tool lr2021_lab ARCHIVE DESTINATION lib RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(FILES include/lriq.h include/lr2021_iq.h include/lr2021_iq_queue.h DESTINATION include)
