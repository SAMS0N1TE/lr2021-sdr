cmake_minimum_required(VERSION 3.18)
project(lr2021_sdr VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
add_library(lriq src/lriq.c src/lriq_fifo.c)
add_library(lr2021_iq src/lr2021_iq.c src/lr2021_iq_queue.c src/lr2021_iq_dev.c src/lr2021_iq_tx.c)
add_library(lr2021_sim src/lr2021_sim.c)
target_link_libraries(lr2021_sim PUBLIC lr2021_iq)
add_library(lr2021_bridge src/lr2021_bridge.c src/lr2021_bridge_server.c)
target_link_libraries(lr2021_bridge PUBLIC lr2021_iq)
foreach(target lriq lr2021_iq lr2021_sim lr2021_bridge)
  target_include_directories(${target} PUBLIC include)
  if(NOT MSVC)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
endforeach()
if(NOT MSVC)
  target_link_libraries(lriq PUBLIC m)
  target_link_libraries(lr2021_sim PUBLIC m)
endif()
add_executable(lriq_tool tools/lriq_tool.c)
add_executable(lr2021_lab tools/lr2021_lab.c)
target_link_libraries(lriq_tool PRIVATE lriq)
if(NOT MSVC)
  target_compile_options(lriq_tool PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(lr2021_lab PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(lr2021_lab PRIVATE m)
endif()

# Host side (Linux, macOS, Windows with MinGW-w64): bridge client, in-process
# server, simulator tool, local spidev/libgpiod backend (Linux) and the
# SoapySDR module (Linux, macOS).
if(UNIX OR MINGW)
  add_library(lr2021_host src/host/host.c src/host/host_server.c src/host/host_spidev.c)
  if(WIN32)
    target_sources(lr2021_host PRIVATE src/host/host_win32.c)
    target_link_libraries(lr2021_host PUBLIC lr2021_bridge lr2021_sim ws2_32)
  else()
    find_package(Threads REQUIRED)
    target_sources(lr2021_host PRIVATE src/host/host_io.c src/host/host_linux_baud.c)
    target_link_libraries(lr2021_host PUBLIC lr2021_bridge lr2021_sim Threads::Threads)
  endif()
  option(LR2021_WITH_GPIOD "Build the local spidev/libgpiod backend (Linux)" ON)
  if(LR2021_WITH_GPIOD AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_path(GPIOD_INCLUDE_DIR gpiod.h)
    find_library(GPIOD_LIBRARY gpiod)
    if(GPIOD_INCLUDE_DIR AND GPIOD_LIBRARY)
      include(CheckSymbolExists)
      set(CMAKE_REQUIRED_INCLUDES ${GPIOD_INCLUDE_DIR})
      check_symbol_exists(gpiod_chip_request_lines gpiod.h LR2021_GPIOD_V2)
      target_include_directories(lr2021_host PRIVATE ${GPIOD_INCLUDE_DIR})
      target_link_libraries(lr2021_host PRIVATE ${GPIOD_LIBRARY})
      target_compile_definitions(lr2021_host PRIVATE LR2021_HAVE_GPIOD
        $<$<BOOL:${LR2021_GPIOD_V2}>:LR2021_GPIOD_V2>)
      message(STATUS "lr2021: local spidev backend with libgpiod ${GPIOD_LIBRARY}")
    else()
      message(STATUS "lr2021: libgpiod not found; local spidev backend disabled")
    endif()
  endif()
  add_executable(lr2021_bridge_sim tools/lr2021_bridge_sim.c)
  target_link_libraries(lr2021_bridge_sim PRIVATE lr2021_host)
  # rtl_tcp-compatible server: the library part is shared with its test.
  add_library(lr2021_tcp_server STATIC tools/lr2021_tcp_server.c)
  target_include_directories(lr2021_tcp_server PUBLIC tools)
  target_link_libraries(lr2021_tcp_server PUBLIC lr2021_host)
  add_executable(lr2021_tcp tools/lr2021_tcp.c)
  target_link_libraries(lr2021_tcp PRIVATE lr2021_tcp_server)
  foreach(target lr2021_host lr2021_bridge_sim lr2021_tcp_server lr2021_tcp)
    target_compile_options(${target} PRIVATE -Wall -Wextra $<$<PLATFORM_ID:Linux>:-Werror>)
  endforeach()
  if(BUILD_TESTING)
    add_executable(test_host tests/test_host.c)
    target_link_libraries(test_host PRIVATE lr2021_host)
    target_compile_options(test_host PRIVATE -Wall -Wextra $<$<PLATFORM_ID:Linux>:-Werror>)
    add_test(NAME host_client COMMAND test_host)
    set_tests_properties(host_client PROPERTIES TIMEOUT 60)
    add_executable(test_rtl_tcp tests/test_rtl_tcp.c)
    target_link_libraries(test_rtl_tcp PRIVATE lr2021_tcp_server)
    target_compile_options(test_rtl_tcp PRIVATE -Wall -Wextra $<$<PLATFORM_ID:Linux>:-Werror>)
    add_test(NAME rtl_tcp_server COMMAND test_rtl_tcp)
    set_tests_properties(rtl_tcp_server PROPERTIES TIMEOUT 60)
    if(Python3_Interpreter_FOUND)
      add_test(NAME rtl_tcp_client COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_rtl_tcp_client.py $<TARGET_FILE:lr2021_tcp>)
      set_tests_properties(rtl_tcp_client PROPERTIES TIMEOUT 60)
      add_test(NAME agc_readback_tools COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_agc_tools.py $<TARGET_FILE:lr2021_bridge_sim>)
      set_tests_properties(agc_readback_tools PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
    endif()
  endif()
  option(LR2021_WITH_SOAPY "Build the SoapySDR module when SoapySDR is found" ON)
  if(LR2021_WITH_SOAPY AND UNIX)
    find_package(SoapySDR CONFIG QUIET)
    if(SoapySDR_FOUND)
      add_subdirectory(soapy)
    else()
      message(STATUS "lr2021: SoapySDR not found; module not built")
    endif()
  endif()
  install(TARGETS lr2021_host lr2021_bridge_sim lr2021_tcp ARCHIVE DESTINATION lib RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
  install(FILES include/lr2021_host.h DESTINATION include)
endif()
